**************criar submodule local
cd ./projeto1
git init
git remote add origin git@github.com:oscarlmjr/JS_Section_15_api_rest.git
git remote set-url origin git@github.com:oscarlmjr/JS_Section_15_api_rest.git
git status > git add . > git commit -m 'Adicionando arquivos e submodule'
git push origin main >> --force
git pull origin main >> --allow-unrelated-histories
cd ..
git submodule add git@github.com:oscarlmjr/JS_Section_15_api_rest.git Section_15_api_rest 
git submodule status
git submodule set-url Section_15_api_rest git@github.com:oscarlmjr/JS_Section_15_api_rest.git
git submodule sync --recursive
git status > git add .gitmodules > git commit -m 'Adicionando arquivos e submodule'
git push --set-upstream origin main
